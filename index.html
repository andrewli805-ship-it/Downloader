function calculateEfficiency() {

    const inputVal = parseFloat(document.getElementById('inputE').value);

    const outputVal = parseFloat(document.getElementById('outputE').value);


    // Validation

    if (isNaN(inputVal) || isNaN(outputVal) || inputVal <= 0) {

        showFeedback("Please enter valid positive numbers. Input must be greater than 0.");

        return;

    }


    if (outputVal > inputVal) {

        showFeedback("Wait! Useful output cannot exceed input energy (Violates Conservation Law).");

        return;

    }


    // UI State Switching

    document.getElementById('result-initial').classList.add('hidden');

    document.getElementById('result-active').classList.remove('hidden');


    // Calculations

    const efficiency = (outputVal / inputVal) * 100;

    const wasted = inputVal - outputVal;


    // Update Text and UI

    document.getElementById('efficiency-val').innerText = efficiency.toFixed(1) + "%";

    document.getElementById('efficiency-bar').style.width = efficiency + "%";

    document.getElementById('wasted-val').innerText = wasted.toFixed(1) + " J";


    // Dynamic Grading Logic (Simple Simulation)

    const gradeBox = document.getElementById('grade-box');

    if (efficiency >= 90) {

        gradeBox.innerText = "Grade 1";

        gradeBox.className = "p-4 rounded-xl shadow-sm bg-green-600 text-white font-bold flex items-center justify-center";

    } else if (efficiency >= 80) {

        gradeBox.innerText = "Grade 2";

        gradeBox.className = "p-4 rounded-xl shadow-sm bg-green-400 text-white font-bold flex items-center justify-center";

    } else if (efficiency >= 70) {

        gradeBox.innerText = "Grade 3";

        gradeBox.className = "p-4 rounded-xl shadow-sm bg-yellow-400 text-white font-bold flex items-center justify-center";

    } else if (efficiency >= 60) {

        gradeBox.innerText = "Grade 4";

        gradeBox.className = "p-4 rounded-xl shadow-sm bg-orange-400 text-white font-bold flex items-center justify-center";

    } else {

        gradeBox.innerText = "Grade 5";

        gradeBox.className = "p-4 rounded-xl shadow-sm bg-red-500 text-white font-bold flex items-center justify-center";

    }

}


function showFeedback(msg) {

    const box = document.getElementById('calc-result');

    const originalContent = box.innerHTML;

    box.innerHTML = `<div class='text-red-500 font-bold p-4 bg-red-50 rounded-xl'>${msg}</div>`;

    

    setTimeout(() => {

        box.innerHTML = originalContent;

    }, 3000);

}

function calculateEfficiency() {

    const inputVal = parseFloat(document.getElementById('inputE').value);

    const outputVal = parseFloat(document.getElementById('outputE').value);


    // Validation

    if (isNaN(inputVal) || isNaN(outputVal) || inputVal <= 0) {

        showFeedback("Please enter valid positive numbers. Input must be greater than 0.");

        return;

    }


    if (outputVal > inputVal) {

        showFeedback("Wait! Useful output cannot exceed input energy (Violates Conservation Law).");

        return;

    }


    // UI State Switching

    document.getElementById('result-initial').classList.add('hidden');

    document.getElementById('result-active').classList.remove('hidden');


    // Calculations

    const efficiency = (outputVal / inputVal) * 100;

    const wasted = inputVal - outputVal;


    // Update Text and UI

    document.getElementById('efficiency-val').innerText = efficiency.toFixed(1) + "%";

    document.getElementById('efficiency-bar').style.width = efficiency + "%";

    document.getElementById('wasted-val').innerText = wasted.toFixed(1) + " J";


    // Dynamic Grading Logic (Simple Simulation)

    const gradeBox = document.getElementById('grade-box');

    if (efficiency >= 90) {

        gradeBox.innerText = "Grade 1";

        gradeBox.className = "p-4 rounded-xl shadow-sm bg-green-600 text-white font-bold flex items-center justify-center";

    } else if (efficiency >= 80) {

        gradeBox.innerText = "Grade 2";

        gradeBox.className = "p-4 rounded-xl shadow-sm bg-green-400 text-white font-bold flex items-center justify-center";

    } else if (efficiency >= 70) {

        gradeBox.innerText = "Grade 3";

        gradeBox.className = "p-4 rounded-xl shadow-sm bg-yellow-400 text-white font-bold flex items-center justify-center";

    } else if (efficiency >= 60) {

        gradeBox.innerText = "Grade 4";

        gradeBox.className = "p-4 rounded-xl shadow-sm bg-orange-400 text-white font-bold flex items-center justify-center";

    } else {

        gradeBox.innerText = "Grade 5";

        gradeBox.className = "p-4 rounded-xl shadow-sm bg-red-500 text-white font-bold flex items-center justify-center";

    }

}


function showFeedback(msg) {

    const box = document.getElementById('calc-result');

    const originalContent = box.innerHTML;

    box.innerHTML = `<div class='text-red-500 font-bold p-4 bg-red-50 rounded-xl'>${msg}</div>`;

    

    setTimeout(() => {

        box.innerHTML = originalContent;

    }, 3000);

}
